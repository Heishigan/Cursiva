{%- if not ns.first %}\cvsep {% endif -%}

{%- elif key in ['github', 'linkedin', 'portfolio'] -%}\href{ {{- val -}} }{ {{- val|replace('https://', '')|replace('http://', '')|replace('www.', '') -}} }
{%- else -%}\mbox{ {{- val -}} }{%- endif -%}
{%- set ns.first = false -%}
{%- endif -%}
{%- endfor -%}
}{ {{- parts[0] -}} }{ {{- parts[1] if parts|length > 1 else '' -}} }
