{#- The name and contact line, shared by the CV and the cover letter. Each contact
    detail is kept whole; the line may only wrap between details,
    and the separator is dropped where it wraps (\cvsep is a \discretionary). -#}
%% Centred contact line. If it has to wrap, the finite stretch on both sides
%% makes TeX balance the two lines instead of leaving one detail alone.
\providecommand{\cvcontactline}[1]{\begingroup\leftskip=0pt plus 0.5\linewidth\rightskip=0pt plus 0.5\linewidth\parfillskip=0pt\parindent=0pt\noindent #1\par\endgroup}
{%- set full = personal_info.get('name', '') -%}
{%- set parts = full.split(' ', 1) -%}
\cvheader{ {{- full -}} }{\cvcontactline{
{%- set ns = namespace(first=true) -%}
{%- for key in ['location', 'phone', 'email', 'github', 'linkedin', 'portfolio'] -%}
{%- set val = personal_info.get(key, '') -%}
{%- if val -%}
{%- if not ns.first %}\cvsep {% endif -%}
{%- if key == 'email' -%}\href{mailto:{{ val }}}{ {{- val -}} }
{%- elif key in ['github', 'linkedin', 'portfolio'] -%}\href{ {{- val -}} }{ {{- val|replace('https://', '')|replace('http://', '')|replace('www.', '') -}} }
{%- else -%}\mbox{ {{- val -}} }{%- endif -%}
{%- set ns.first = false -%}
{%- endif -%}
{%- endfor -%}
}}{ {{- parts[0] -}} }{ {{- parts[1] if parts|length > 1 else '' -}} }
